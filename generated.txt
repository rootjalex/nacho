template<typename index_t, typename value_t>
__global__ 
void partition_ij_kernel(const a_tensor_format<index_t, value_t> a, const b_tensor_format<index_t, value_t> b, partition_ij<index_t> partitions, const index_t per_thread_work) {
  int32_t thread_id = (blockIdx.x * blockDim.x) + threadIdx.x;
  index_t count = thread_id * per_thread_work;
  if (count == 0) {
    partitions.a_i_p[thread_id] = 0;
    partitions.b_i_p[thread_id] = 0;
    partitions.a_j_p[thread_id] = -1;
    partitions.b_j_p[thread_id] = -1;
    return;
  }
  if (((0 + a.nnz) + b.nnz) <= count) {
    partitions.a_i_p[thread_id] = a.dim_i_length - 1;
    partitions.b_i_p[thread_id] = b.dim_i_length - 1;
    partitions.a_j_p[thread_id] = a.dim_j_length - 1;
    partitions.b_j_p[thread_id] = b.dim_j_length - 1;
    return;
  }
  index_t rem_count = count;
  index_t total_work = 0;
  bool is_a_partitioned = 0;
  bool is_b_partitioned = 0;
  index_t start_i = -1;
  index_t end_i = a.dim_i_size - 1;
  index_t mid_i = start_i + (((end_i - start_i) + 1) / 2);
  index_t a_i_p_start = -1;
  index_t a_i_p_end = a.dim_i_length - 1;
  index_t a_i_p = a_i_p_end;
  index_t b_i_p_start = -1;
  index_t b_i_p_end = b.dim_i_length - 1;
  index_t b_i_p = b_i_p_end;
  while (1) {
    mid_i = start_i + (((end_i - start_i) + 1) / 2);
    a_i_p = binary_search(a.dim_i_indices, mid_i - 1, a_i_p_start, a_i_p_end);
    b_i_p = binary_search(b.dim_i_indices, mid_i - 1, b_i_p_start, b_i_p_end);
    total_work = 0;
    total_work = total_work + work_ij_a_dim_i(a, a_i_p);
    total_work = total_work + work_ij_b_dim_i(b, b_i_p);
    if (end_i <= start_i) {
      a_i_p = binary_search(a.dim_i_indices, mid_i, a_i_p_start, a_i_p_end);
      partitions.a_i_p[thread_id] = a_i_p;
      is_a_partitioned = is_a_partitioned || (a.dim_i_indices[a_i_p] != mid_i);
      b_i_p = binary_search(b.dim_i_indices, mid_i, b_i_p_start, b_i_p_end);
      partitions.b_i_p[thread_id] = b_i_p;
      is_b_partitioned = is_b_partitioned || (b.dim_i_indices[b_i_p] != mid_i);
      rem_count = rem_count - total_work;
      break;
    }
    if (total_work < rem_count) {
      start_i = mid_i;
    } else {
      end_i = mid_i - 1;
    }
  }
  index_t start_j = -1;
  index_t end_j = a.dim_j_size - 1;
  index_t mid_j = start_j + ((end_j - start_j) / 2);
  index_t a_j_p_start = a.dim_j_offsets[a_i_p] - 1;
  index_t a_j_p_end = a.dim_j_offsets[a_i_p + 1] - 1;
  index_t a_j_p = a_j_p_end;
  index_t b_j_p_start = b.dim_j_offsets[b_i_p] - 1;
  index_t b_j_p_end = b.dim_j_offsets[b_i_p + 1] - 1;
  index_t b_j_p = b_j_p_end;
  while (1) {
    mid_j = start_j + ((end_j - start_j) / 2);
    if (!is_a_partitioned) {
      a_j_p = binary_search(a.dim_j_indices, mid_j, a_j_p_start, a_j_p_end);
    }
    if (!is_b_partitioned) {
      b_j_p = binary_search(b.dim_j_indices, mid_j, b_j_p_start, b_j_p_end);
    }
    total_work = 0;
    total_work = total_work + (is_a_partitioned ? 0 : work_ij_a_dim_j(a, a_i_p, a_j_p));
    total_work = total_work + (is_b_partitioned ? 0 : work_ij_b_dim_j(b, b_i_p, b_j_p));
    if (end_j <= start_j) {
      partitions.a_j_p[thread_id] = a_j_p;
      partitions.b_j_p[thread_id] = b_j_p;
      rem_count = rem_count - total_work;
      break;
    }
    if (total_work < rem_count) {
      start_j = mid_j + 1;
    } else {
      end_j = mid_j;
    }
  }
  return;
